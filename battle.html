import re from openpyxl.formatting.rule import FormulaRule from openpyxl.styles import Alignment, Border, Font, PatternFill, Side import pandas as pd def estrai_somma_note(testo): """Estrae tutti i numeri da una stringa di testo e ne calcola la somma.""" if pd.isna(testo) or not str(testo).strip(): return None numeri = re.findall(r"\b\d+\b", str(testo)) return sum(map(int, numeri)) if numeri else None def automatizza_magazzino(file_input, file_output): # 1. Carica il file grezzo saltando la riga 1 ("Brogliaccio documenti") df = pd.read_excel(file_input, skiprows=1) # 2. Rimuovi la colonna 'UM' (che contiene 'NR') if "UM" in df.columns: df = df.drop(columns=["UM"]) # 3. Calcola in automatico la colonna 'Contate' (ex colonna G, ora F) sommando i numeri nelle Note df["Contate"] = df["Note"].apply(estrai_somma_note) # Se la nota è vuota ma c'è quantità video > 0, possiamo considerare l'articolo allineato (opzionale) df["Contate"] = df.apply( lambda r: ( r["Quantità video"] if pd.isna(r["Contate"]) and r["Quantità video"] > 0 and pd.isna(r["Note"]) else r["Contate"] ), axis=1, ) # 4. Calcola la colonna di controllo della Differenza df["Differenza"] = df["Contate"] - df["Quantità video"] # 5. Esporta in Excel tramite openpyxl per applicare gli stili with pd.ExcelWriter(file_output, engine="openpyxl") as writer: df.to_excel(writer, sheet_name="Controllo_Magazzino", index=False) ws = writer.sheets["Controllo_Magazzino"] # --- FORMATTAZIONE CONDIZIONALE --- max_row = len(df) + 1 # Colore Grigio per Quantità Video == 0 grigio_fill = PatternFill( start_color="D5D8DC", end_color="D5D8DC", fill_type="solid" ) ws.conditional_formatting.add( f"A2:I{max_row}", FormulaRule(formula=["$E2=0"], stopIfTrue=True, fill=grigio_fill), ) # Colore Rosso Chiaro per Ammanco (Differenza < 0 -> Video > Contate) rosso_fill = PatternFill( start_color="FADBD8", end_color="FADBD8", fill_type="solid" ) ws.conditional_formatting.add( f"A2:I{max_row}", FormulaRule( formula=["AND($H2<0, $H2<>'')"], stopIfTrue=True, fill=rosso_fill ), ) # Colore Verde Chiaro per Eccedenza (Differenza > 0 -> Video < Contate) verde_fill = PatternFill( start_color="D4EFDF", end_color="D4EFDF", fill_type="solid" ) ws.conditional_formatting.add( f"A2:I{max_row}", FormulaRule( formula=["AND($H2>0, $H2<>'')"], stopIfTrue=True, fill=verde_fill ), ) # Auto-adattamento larghezza colonne for col in ws.columns: max_len = max(len(str(cell.value or "")) for cell in col) col_letter = col[0].column_letter ws.column_dimensions[col_letter].width = max(max_len + 3, 12) print( f"File elaborato con successo e salvato in: {file_output}" ) # Esempio di utilizzo: # automatizza_magazzino("export_grezzo.xlsx", "Magazzino_Controllato.xlsx")
